# Explore features #######
model2 <- glm(outcome ~ num_contesters + contested + gamestate + shottype+ distance + three + dribblesbefore + shotclock + locationx + closestdefdist + shooterspeed, data = train_df, family = 'binomial')
summary(model2)


# Prepare Model #######
preds2 <-predict(model2, newdata = test_df, type = 'response')

# Validating model #######
summary(preds2)
mean(train_df$outcome) # comparing mean

set.seed(42)  # makes the random split reproducible
train_indices <- sample(1:nrow(train_df), size = 0.8 * nrow(train_df))

train_split <- train_df[train_indices, ]   # 80% — fit the model on this
val_split <- train_df[-train_indices, ]    # 20% — check performance on this

model_val <- glm(outcome ~ num_contesters + contested + gamestate + shottype + distance + three + dribblesbefore + shotclock + locationx + closestdefdist + shooterspeed,
                 data = train_split, family = 'binomial')

val_preds <- predict(model_val, newdata = val_split, type = "response")

log_loss <- function(actual, predicted) {
  eps <- 1e-15
  predicted <- pmin(pmax(predicted, eps), 1 - eps)
  -mean(actual * log(predicted) + (1 - actual) * log(1 - predicted))
}

log_loss(as.numeric(val_split$outcome), val_preds)

# Baseline: predicting the mean outcome for every validation shot #######
baseline_pred <- mean(train_split$outcome)   # 0.4580698 — the mean from your training split
baseline_preds <- rep(baseline_pred, nrow(val_split))

log_loss(as.numeric(val_split$outcome), baseline_preds)
